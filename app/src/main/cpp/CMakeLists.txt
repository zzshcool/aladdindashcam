cmake_minimum_required(VERSION 3.22.1)
project(dashcam_export)
add_library(oem_ffmpeg SHARED IMPORTED)
set_target_properties(oem_ffmpeg PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libffmpeg.so")
add_library(dashcam_export SHARED video_export.cpp)
target_include_directories(dashcam_export PRIVATE ffmpeg-4.0.4)
target_compile_features(dashcam_export PRIVATE cxx_std_17)
target_compile_options(dashcam_export PRIVATE -Wno-deprecated-declarations)
target_link_options(dashcam_export PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(dashcam_export oem_ffmpeg jnigraphics log)
